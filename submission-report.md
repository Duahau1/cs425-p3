## Design

The implementation is organized around the three parts of the project: address and CIDR handling, subnet arithmetic, and forwarding-table lookup. The public types and function declarations are in `src/lab.h`; the implementation is in `src/lab.c`. IPv4 addresses are stored in a `uint32_t` with the first octet in the most significant byte, which keeps parsing, formatting, masking, and comparisons consistent.

Address and CIDR parsing validate input before writing to the caller's output. Decimal values are parsed digit by digit, rejecting malformed values, leading zeroes, and values above the permitted range. Subnet operations use a mask derived from the prefix. Host-count and host-address functions handle `/31` and `/32` separately, avoiding the usual network and broadcast assumptions for those prefixes.

The forwarding table is opaque to callers and stores routes in a dynamically growing array. Route parsing validates the destination, gateway, and device fields; table loading removes comments, skips blank lines, and stops at the first invalid route while reporting its line number. Lookup scans matching routes and returns the one with the longest prefix, so more specific routes take precedence over a default route.

```c
static bool is_space_character(char inputChar)
{
  return inputChar == ' ' || inputChar == '\t' || inputChar == '\n' || inputChar == '\r' || inputChar == '\v' || inputChar == '\f';
}

static bool is_digit_character(char digitChar)
{
  return digitChar >= '0' && digitChar <= '9';
}

uint8_t bound_prefix(uint8_t prefix)
{
  return prefix > MAXIMUM_DECIMAL_PREFIX_VALUE ? MAXIMUM_DECIMAL_PREFIX_VALUE : prefix;
}

static bool append_digit_to_octet(char digit_char, uint32_t max, uint32_t *octet_value)
{

  uint32_t digit = (uint32_t)(digit_char - '0');

  if (*octet_value > (max - digit) / BASE_10)
  {
    return false;
  }

  *octet_value = (*octet_value * BASE_10) + digit;
  return true;
}

static const char *parse_decimal(const char *cursor, uint32_t max, uint32_t *value)
{
  if (!is_digit_character(*cursor))
  {
    return NULL;
  }

  if (cursor[0] == '0' && is_digit_character(cursor[1]))
  {
    return NULL;
  }

  uint32_t parsed_value = 0;

  while (is_digit_character(*cursor))
  {
    if (!append_digit_to_octet(*cursor, max, &parsed_value))
    {
      return NULL;
    }
    cursor++;
  }
  *value = parsed_value;
  return cursor;
}

/**
 * Scan the next whitespace-separated token from the range [cursor, end) into
 * the destination buffer (which must hold up to `buffer_size` bytes).
 *
 * Returns a pointer to the character immediately following the token,
 * or NULL if the token is missing, empty, or exceeds the buffer size.
 */
static const char *scan_next_token(const char *cursor, const char *end, char *buffer, size_t buffer_size)
{
  while (cursor < end && (*cursor == ' ' || *cursor == '\t'))
  {
    cursor++;
  }

  if (cursor >= end)
  {
    return NULL;
  }

  const char *token_start = cursor;
  while (cursor < end && *cursor != ' ' && *cursor != '\t')
  {
    cursor++;
  }

  size_t token_length = (size_t)(cursor - token_start);

  if (token_length == 0 || token_length >= buffer_size)
  {
    return NULL;
  }

  memcpy(buffer, token_start, token_length);
  buffer[token_length] = '\0';

  return cursor;
}

bool ip_parse(const char *text, uint32_t *addr)
{
  if (text == NULL || addr == NULL)
  {
    return false;
  }

  uint32_t result = 0;

  for (int i = 0; i < 4; i++)
  {
    if (i > 0)
    {
      if (*text != '.')
      {
        return false;
      }
      text++;
    }

    unsigned int octet;
    text = parse_decimal(text, MAXIMUM_DECIMAL_VALUE_PER_OCTET, &octet);
    if (text == NULL)
    {
      return false;
    }

    result = (result << 8) | octet;
  }

  if (*text != '\0')
  {
    return false;
  }

  *addr = result;
  return true;
}

void ip_format(uint32_t addr, char out[IP_STR_LEN])
{
  snprintf(out, IP_STR_LEN, "%u.%u.%u.%u",
           (unsigned int)((addr >> 24) & MAXIMUM_DECIMAL_VALUE_PER_OCTET),

           (unsigned int)((addr >> 8) & MAXIMUM_DECIMAL_VALUE_PER_OCTET),
           (unsigned int)(addr & MAXIMUM_DECIMAL_VALUE_PER_OCTET));
}

bool cidr_parse(const char *text, cidr_t *block)
{
  if (text == NULL || block == NULL)
  {
    return false;
  }
  const char *slash = strchr(text, '/');
  if (slash == NULL || strchr(slash + 1, '/') != NULL)
  {
    return false;
  }
  // Extract and validate the IP address portion before the slash
  size_t address_len = (size_t)(slash - text);
  if (address_len == 0 || address_len >= IP_STR_LEN)
  {
    return false;
  }

  char address[IP_STR_LEN];
  memcpy(address, text, address_len);
  address[address_len] = '\0';

  cidr_t parsed;
  if (!ip_parse(address, &parsed.addr))
  {
    return false;
  }

  const char *prefix_text = slash + 1;
  uint32_t prefix;

  const char *end = parse_decimal(prefix_text, MAXIMUM_DECIMAL_PREFIX_VALUE, &prefix);
  if (end == NULL || *end != '\0' || prefix > MAXIMUM_DECIMAL_PREFIX_VALUE)
  {
    return false; // Invalid prefix value, parsing error, or trailing characters
  }

  parsed.prefix = (uint8_t)prefix;
  *block = parsed;
  return true;
}

uint32_t cidr_mask(uint8_t prefix)
{
  prefix = bound_prefix(prefix);
  if (prefix == 0)
  {
    return 0;
  }
  return UINT32_MAX << (MAXIMUM_DECIMAL_PREFIX_VALUE - prefix);
}

uint32_t cidr_network(cidr_t block)
{
  return block.addr & cidr_mask(block.prefix);
}

uint32_t cidr_broadcast(cidr_t block)
{
  return block.addr | ~cidr_mask(block.prefix);
}

uint32_t cidr_first_host(cidr_t block)
{
  uint32_t network = cidr_network(block);
  return bound_prefix(block.prefix) >= 31 ? network : network + 1;
}

uint32_t cidr_last_host(cidr_t block)
{
  uint32_t broadcast = cidr_broadcast(block);
  return bound_prefix(block.prefix) >= 31 ? broadcast : broadcast - 1;
}

uint64_t cidr_host_count(cidr_t block)
{
  uint8_t prefix = bound_prefix(block.prefix);
  if (prefix == 32)
  {
    return 1;
  }
  if (prefix == 31)
  {
    return 2;
  }
  return (UINT64_C(1) << (32 - prefix)) - 2;
}

bool cidr_contains(cidr_t block, uint32_t addr)
{
  uint32_t mask = cidr_mask(block.prefix);
  return (addr & mask) == (block.addr & mask);
}

bool cidr_is_network(cidr_t block)
{
  return cidr_network(block) == block.addr;
}

bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out)
{
  if (out == NULL || n == 0 || !cidr_is_network(block) || block.prefix > 32)
  {
    return false;
  }
  // Ensure n is a power of two and the index i is within bounds (0 <= i < n)
  if ((n & (n - 1U)) != 0 || i >= n)
  {
    return false;
  }

  uint32_t bits = 0;
  while ((UINT32_C(1) << bits) < n)
  {
    bits++;
  }
  uint32_t prefix = bound_prefix(block.prefix) + bits;
  if (prefix > 32 || i >= n)
  {
    return false;
  }
  uint64_t subnet_size = UINT64_C(1) << (32U - prefix);
  *out = (cidr_t){
      .addr = (uint32_t)((uint64_t)block.addr + (uint64_t)i * subnet_size),
      .prefix = (unsigned char)prefix};

  return true;
}

bool route_parse(const char *line, route_t *route)
{

  if (line == NULL || route == NULL)
  {
    return false;
  }

  const char *start = line;
  while (is_space_character(*start))
  {
    start++;
  }

  const char *end = start + strlen(start);
  while (end > start && is_space_character(end[-1]))
  {
    end--;
  }

  char dest_str[CIDR_STR_LEN];
  char gateway_str[IP_STR_LEN];
  char dev_str[DEV_NAME_LEN];

  const char *p = scan_next_token(start, end, dest_str, sizeof(dest_str));
  if (p != NULL)
  {
    p = scan_next_token(p, end, gateway_str, sizeof(gateway_str));
  }
  if (p != NULL)
  {
    p = scan_next_token(p, end, dev_str, sizeof(dev_str));
  }

  if (p == NULL || p != end)
  {
    return false;
  }

  route_t parsed;
  memset(&parsed, 0, sizeof(parsed));

  if (strcmp(dest_str, "default") == 0)
  {
    parsed.dest.addr = 0;
    parsed.dest.prefix = 0;
  }
  else
  {
    if (!cidr_parse(dest_str, &parsed.dest) || !cidr_is_network(parsed.dest))
    {
      return false;
    }
  }

  // 2. Parse Gateway (IPv4 address or "direct")
  if (strcmp(gateway_str, "direct") == 0)
  {
    parsed.gateway = 0;
  }
  else
  {
    if (!ip_parse(gateway_str, &parsed.gateway))
    {
      return false;
    }
  }

  for (const char *c = dev_str; *c != '\0'; c++)
  {
    if (is_space_character(*c))
    {
      return false;
    }
  }
  memcpy(parsed.dev, dev_str, strlen(dev_str) + 1);

  *route = parsed;
  return true;
}

bool table_add(fwd_table_t *table, const route_t *route)
{
  if (table == NULL || route == NULL || !cidr_is_network(route->dest))
  {
    return false;
  }

  // Check for duplicate routes using normalized prefix comparison
  uint8_t normalized_prefix = bound_prefix(route->dest.prefix);
  for (size_t i = 0; i < table->count; ++i)
  {
    const cidr_t *existing_dest = &table->routes[i].dest;
    if (existing_dest->addr == route->dest.addr && bound_prefix(existing_dest->prefix) == normalized_prefix)
    {
      return false; // Route already exists
    }
  }

  if (table->count == table->capacity)
  {
    size_t new_capacity = table->capacity == 0 ? 8U : table->capacity * 2U;
    route_t *grown_routes = realloc(table->routes, new_capacity * sizeof(*grown_routes));

    if (grown_routes == NULL)
    {               // GCOVR_EXCL_START
      return false; // realloc failed, original table remains intact and valid
    } // GCOVR_EXCL_STOP

    table->routes = grown_routes;
    table->capacity = new_capacity;
  }

  route_t *slot = &table->routes[table->count++];
  *slot = *route;
  slot->dev[DEV_NAME_LEN - 1] = '\0';

  return true;
}

bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line)
{
  if (bad_line != NULL)
  {
    *bad_line = 0;
  }
  if (table == NULL || in == NULL)
  {
    return false;
  }

  char *line_buffer = NULL;
  size_t buffer_capacity = 0;
  size_t current_line_number = 0;
  bool load_successful = true;

  while (getline(&line_buffer, &buffer_capacity, in) != -1)
  {
    current_line_number++;

    char *comment_marker = strchr(line_buffer, '#');
    if (comment_marker != NULL)
    {
      *comment_marker = '\0';
    }

    const char *cursor = line_buffer;
    while (is_space_character(*cursor))
    {
      cursor++;
    }
    if (*cursor == '\0')
    {
      continue;
    }

    route_t parsed_route;
    if (!route_parse(line_buffer, &parsed_route) || !table_add(table, &parsed_route))
    {
      if (bad_line != NULL)
      {
        *bad_line = current_line_number;
      }
      load_successful = false;
      break;
    }
  }

  // Check for stream reading errors (keeping bad_line as 0 if file I/O failed)
  if (load_successful && ferror(in))
  {
    load_successful = false;
  }

  free(line_buffer);
  return load_successful;
}

  if (table == NULL)
  {
    return NULL;
  }

  const route_t *best_match = NULL;

  for (size_t i = 0; i < table->count; ++i)
  {
    const route_t *candidate_route = &table->routes[i];

    if (cidr_contains(candidate_route->dest, dest))
    {
      if (best_match == NULL ||
          bound_prefix(candidate_route->dest.prefix) > bound_prefix(best_match->dest.prefix))
      {
        best_match = candidate_route;
      }
    }
  }

  return best_match;
}
```

```c

#ifndef LAB_H
#define LAB_H

#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>

/*
 * Every address in this file is a uint32_t in host byte order, so the first
 * octet of 192.168.1.10 is the most significant byte: 0xC0A8010A.
 */

/** Room for the longest dotted quad, "255.255.255.255", and its '\0'. */
#define IP_STR_LEN 16

/** Room for the longest CIDR string, "255.255.255.255/32", and its '\0'. */
#define CIDR_STR_LEN 19

/** Room for an interface name; IFNAMSIZ on Linux, including the '\0'. */
#define DEV_NAME_LEN 16

#define MAXIMUM_DECIMAL_VALUE_PER_OCTET 255U

#define MAXIMUM_DECIMAL_PREFIX_VALUE 32U

#define BASE_10 10U

/* Exit codes */
#define EXIT_BAD_CMD_OR_MALFORMED 1
#define EXIT_UNOPENED_TABLE_OR_BAD_LINE 2
#define EXIT_NO_ROUTE_MATCH 3

/** An address block in CIDR notation: an address and a prefix length. */
typedef struct
{
    uint32_t addr;  /**< any address in the block, host byte order */
    uint8_t prefix; /**< 0 to 32 */
} cidr_t;

/** One entry in a forwarding table. */
typedef struct
{
    cidr_t dest;            /**< destination block; host bits must be zero */
    uint32_t gateway;       /**< next hop, or 0 when the block is directly connected */
    char dev[DEV_NAME_LEN]; /**< outgoing interface, for example "eth0" */
} route_t;

/** A forwarding table. Its layout is private to lab.c. */
typedef struct fwd_table fwd_table_t;

uint8_t bound_prefix(uint8_t prefix);

/* ------------------------------------------------------------------------
 * Task 1 - Address parsing and formatting
 * --------------------------------------------------------------------- */

/**
 * @brief Parse a dotted quad such as "192.168.1.10".
 *
 * Exactly four decimal octets from 0 to 255 separated by single periods, with
 * nothing before, between or after them. An octet with a leading zero ("010")
 * is rejected, because inet_aton() reads it as octal and that is never what a
 * person typing it meant.
 *
 * @param text The string to parse.
 * @param addr Where the address is stored on success; untouched on failure.
 * @return true if text is a valid dotted quad, false otherwise (including NULL).
 */
bool ip_parse(const char *text, uint32_t *addr);

/**
 * @brief Format an address as a dotted quad.
 * @param addr The address, host byte order.
 * @param out A buffer of at least IP_STR_LEN bytes.
 */
void ip_format(uint32_t addr, char out[IP_STR_LEN]);

/**
 * @brief Parse a block in CIDR notation such as "192.168.1.0/24".
 *
 * The address follows the rules of ip_parse() and the prefix is a decimal
 * number from 0 to 32 with no leading zero. The slash and the prefix are
 * required. Host bits may be set ("132.178.227.11/25" is an interface
 * address), so use cidr_is_network() when they must not be.
 *
 * @param text The string to parse.
 * @param block Where the block is stored on success; untouched on failure.
 * @return true if text is valid CIDR notation, false otherwise (including NULL).
 */
bool cidr_parse(const char *text, cidr_t *block);

/**
 * @brief Format a block as "a.b.c.d/p", using the address exactly as stored.
 *
 * A prefix over 32 is written as 32, the way cidr_mask() treats it.
 *
 * @param block The block.
 * @param out A buffer of at least CIDR_STR_LEN bytes.
 */
void cidr_format(cidr_t block, char out[CIDR_STR_LEN]);

/* ------------------------------------------------------------------------
 * Task 2 - Subnet arithmetic
 * --------------------------------------------------------------------- */

/**
 * @brief The subnet mask for a prefix length: /24 is 255.255.255.0.
 *
 * Careful with /0. Shifting a 32 bit value by 32 is undefined behavior in C.
 *
 * @param prefix 0 to 32; anything larger is treated as 32.
 * @return The mask, host byte order.
 */
uint32_t cidr_mask(uint8_t prefix);

/** @brief The lowest address in the block: the address with the host bits cleared. */
uint32_t cidr_network(cidr_t block);

/** @brief The highest address in the block: the address with the host bits set. */
uint32_t cidr_broadcast(cidr_t block);

/**
 * @brief The first address that can be given to a host.
 *
 * For /0 to /30 that is the network address plus one. A /31 is a point to
 * point link (RFC 3021) where both addresses are usable, and a /32 is a single
 * host, so for those two the first host is the network address.
 */
uint32_t cidr_first_host(cidr_t block);

/**
 * @brief The last address that can be given to a host.
 *
 * The broadcast address minus one for /0 to /30, and the broadcast address
 * itself for /31 and /32 (see cidr_first_host()).
 */
uint32_t cidr_last_host(cidr_t block);

/**
 * @brief How many addresses in the block can be given to hosts.
 *
 * 2^(32 - prefix) - 2 for /0 to /30, 2 for /31, and 1 for /32. For a /0 the
 * 2^32 in that formula needs 33 bits, more than a uint32_t holds, which is why
 * this returns a uint64_t.
 */
uint64_t cidr_host_count(cidr_t block);

/** @brief true if addr falls inside the block. */
bool cidr_contains(cidr_t block, uint32_t addr);

/** @brief true if the block's host bits are all zero, so it names a network. */
bool cidr_is_network(cidr_t block);

/**
 * @brief The i-th of n equally sized subnets of a block.
 *
 * Splitting a /24 into 4 gives four /26 blocks, and subnet 0 is the lowest.
 * Equal sizes are only possible when n is a power of two, and the new prefix
 * can not pass 32.
 *
 * @param block The block to split; it must be a network (cidr_is_network()).
 * @param n How many subnets, a power of two from 1 up.
 * @param i Which subnet, 0 to n - 1.
 * @param out Where the subnet is stored on success; untouched on failure.
 * @return false if out is NULL, the block is not a network, n is not a power
 *         of two, the split would need a prefix longer than 32, or i is out of
 *         range.
 */
bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out);

/* ------------------------------------------------------------------------
 * Task 3 - Longest prefix match
 * --------------------------------------------------------------------- */

/**
 * @brief Parse one line of a forwarding table file.
 *
 * A line is three fields separated by spaces or tabs:
 *
 *     <destination> <gateway> <device>
 *
 * The destination is a block in CIDR notation with no host bits set, or the
 * word "default" for 0.0.0.0/0. The gateway is a dotted quad, or the word
 * "direct" for a directly connected block (stored as 0). The device is 1 to
 * DEV_NAME_LEN - 1 characters. Leading and trailing whitespace, including a
 * trailing newline, is ignored. Comments are not handled here.
 *
 * @param line The line.
 * @param route Where the route is stored on success; untouched on failure.
 * @return true if the line is a valid route, false otherwise (including NULL).
 */
bool route_parse(const char *line, route_t *route);

/**
 * @brief Create an empty forwarding table.
 * @return The table, which the caller frees with table_destroy(), or NULL if
 *         memory could not be allocated.
 */
fwd_table_t *table_create(void);

/** @brief Free a table and everything in it. NULL is allowed and does nothing. */
void table_destroy(fwd_table_t *table);

/** @brief The number of routes in the table; 0 for NULL. */
size_t table_count(const fwd_table_t *table);

/**
 * @brief Add a copy of a route to the table, growing it as needed.
 * @return false if either argument is NULL, the destination has host bits
 *         set, the table already holds a route to the same block, or memory
 *         could not be allocated. The table is unchanged on failure.
 */
bool table_add(fwd_table_t *table, const route_t *route);

/**
 * @brief Read a forwarding table file and add every route in it.
 *
 * Everything from a '#' to the end of a line is a comment, and a line that is
 * blank once the comment is removed is skipped. Every other line must be a
 * valid route (route_parse()) that table_add() accepts.
 *
 * @param table The table to add to.
 * @param in The open file to read.
 * @param bad_line If not NULL, set to the 1-based number of the first line
 *        that failed, or 0 if the arguments were NULL or the file could not
 *        be read.
 * @return true if every line loaded. On false, routes from the lines before
 *         the bad one stay in the table.
 */
bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line);

/**
 * @brief Longest prefix match: the route whose block contains dest and has the
 *        longest prefix.
 *
 * A default route (0.0.0.0/0) matches everything, so it is chosen only when
 * nothing more specific does. The order the routes were added in does not
 * matter.
 *
 * @return The matching route, which stays owned by the table, or NULL if no
 *         route matches (or table is NULL).
 */
const route_t *table_lookup(const fwd_table_t *table, uint32_t dest);

#endif // LAB_H
```

```c
static void usage(FILE *out)

static int info(const char *arg)
{
    cidr_t block;
    char address[IP_STR_LEN];
    char mask[IP_STR_LEN];
    char network[IP_STR_LEN];
    char broadcast[IP_STR_LEN];
    char first[IP_STR_LEN];
    char last[IP_STR_LEN];

    if (!cidr_parse(arg, &block))
    {
        fprintf(stderr, "myapp: critical failure - '%s': not a valid CIDR block\n", arg);
        return EXIT_BAD_CMD_OR_MALFORMED;
    }

    ip_format(block.addr, address);
    ip_format(cidr_mask(block.prefix), mask);
    ip_format(cidr_network(block), network);
    ip_format(cidr_broadcast(block), broadcast);
    ip_format(cidr_first_host(block), first);
    ip_format(cidr_last_host(block), last);
    (void)printf("address    %s\n", address);
    (void)printf("prefix     /%u\n", bound_prefix(block.prefix));
    (void)printf("netmask    %s\n", mask);
    (void)printf("network    %s\n", network);
    (void)printf("broadcast  %s\n", broadcast);
    (void)printf("first host %s\n", first);
    (void)printf("last host  %s\n", last);
    (void)printf("hosts      %" PRIu64 "\n", cidr_host_count(block));

    return EXIT_SUCCESS;
}

static int lookup(int argc, char **argv)
{
    if (argc < 4)
    {
        fprintf(stderr, "myapp: 'lookup' expects at least 2 arguments (<table-file> <address>...), but got %d\n", argc - 2);
        usage(stderr);
        return EXIT_BAD_CMD_OR_MALFORMED;
    }

    const char *table_path = argv[2];
    FILE *in = fopen(table_path, "r");
    if (in == NULL)
    {
        (void)fprintf(stderr, "myapp: error opening table file '%s': %s\n", table_path, strerror(errno));
        return EXIT_UNOPENED_TABLE_OR_BAD_LINE;
    }

    fwd_table_t *table = table_create();
    if (table == NULL)
    {
        (void)fprintf(stderr, "myapp: critical failure — failed to allocate memory for route table\n");
        fclose(in);
        return EXIT_UNOPENED_TABLE_OR_BAD_LINE;
    }

    size_t bad_line;
    bool loaded = table_load(table, in, &bad_line);
    fclose(in);

    if (!loaded)
    {
        if (bad_line > 0)
        {
            fprintf(stderr, "myapp: critical failure — %s:%zu: route appears to be invalid\n", table_path, bad_line);
        }
        else
        {
            fprintf(stderr, "myapp: critical failure — %s: could not be read\n", table_path);
        }
        table_destroy(table);
        return EXIT_UNOPENED_TABLE_OR_BAD_LINE;
    }

    bool saw_bad_address = false;
    bool saw_unmatched = false;

    // Process all destination addresses provided as command-line arguments
    for (int i = 3; i < argc; ++i)
    {
        uint32_t dest;
        if (!ip_parse(argv[i], &dest))
        {
            fprintf(stderr, "myapp: critical failure — %s: not a valid address\n", argv[i]);
            saw_bad_address = true;
            continue;
        }

        char formatted_addr[IP_STR_LEN];
        ip_format(dest, formatted_addr);

        const route_t *route = table_lookup(table, dest);
        if (route == NULL)
        {
            printf("not able to reach the following address: %s\n", formatted_addr);
            saw_unmatched = true;
        }
        else if (route->gateway == 0)
        {
            printf("%s dev %s\n", formatted_addr, route->dev);
        }
        else
        {
            char gateway_addr[IP_STR_LEN];
            ip_format(route->gateway, gateway_addr);
            printf("%s via %s dev %s\n", formatted_addr, gateway_addr, route->dev);
        }
    }

    table_destroy(table);

    if (saw_bad_address)
    {
        return EXIT_BAD_CMD_OR_MALFORMED;
    }
    return saw_unmatched ? EXIT_NO_ROUTE_MATCH : EXIT_SUCCESS;
}
static int split(const char *block_arg, const char *count_arg)
{
    cidr_t block;
    cidr_t subnet;
    uint32_t count;

    if (!cidr_parse(block_arg, &block))
    {
        fprintf(stderr, "myapp: critical failure - failed to parse '%s' as a valid network identifier\n", block_arg);
        return EXIT_BAD_CMD_OR_MALFORMED;
    }

    if (!cidr_is_network(block))
    {
        char normalized_network[CIDR_STR_LEN];
        cidr_format((cidr_t){cidr_network(block), block.prefix}, normalized_network);
        fprintf(stderr, "myapp: critical failure - network address '%s' contains non-zero host bits (did you mean %s?)\n",
                block_arg, normalized_network);
        return EXIT_BAD_CMD_OR_MALFORMED;
    }

    if (count_arg[0] < '1' || count_arg[0] > '9')
    {
        fprintf(stderr, "myapp: critical failure - split count '%s' must be a power of two from 1 up (e.g., 2, 4, 8, 16)\n", count_arg);
        return EXIT_BAD_CMD_OR_MALFORMED;
    }

    uint64_t v = 0;
    for (const char *p = count_arg; *p != '\0'; p++)
    {
        if (*p < '0' || *p > '9')
        {
            fprintf(stderr, "myapp: critical failure - split count '%s' must be a power of two from 1 up (e.g., 2, 4, 8, 16)\n", count_arg);
            return EXIT_BAD_CMD_OR_MALFORMED;
        }
        v = v * 10 + (uint64_t)(*p - '0');
        if (v > UINT32_MAX)
        {
            fprintf(stderr, "myapp: critical failure - split count '%s' must be a power of two from 1 up (e.g., 2, 4, 8, 16)\n", count_arg);
            return EXIT_BAD_CMD_OR_MALFORMED;
        }
    }

    count = (uint32_t)v;

    if ((count & (count - 1U)) != 0)
    {
        fprintf(stderr, "myapp: critical failure - split count '%s' must be a power of two from 1 up (e.g., 2, 4, 8, 16)\n", count_arg);
        return EXIT_BAD_CMD_OR_MALFORMED;
    }

    for (uint32_t i = 0; i < count; ++i)
    {
        char formatted_subnet[CIDR_STR_LEN];

        if (!cidr_subnet(block, count, i, &subnet))
        {
            fprintf(stderr, "myapp: critical failure - prefix limit exceeded when attempting to split %s into %u parts\n",
                    block_arg, count);
            return EXIT_BAD_CMD_OR_MALFORMED;
        }

        cidr_format(subnet, formatted_subnet);
        printf("%s\n", formatted_subnet);
    }

    return EXIT_SUCCESS;
}

            return EXIT_BAD_CMD_OR_MALFORMED;
```

```c
  TEST_ASSERT_FALSE(cidr_subnet(block, 4, 0, NULL));   // out is NULL

static void test_route_parse(void)
{
  route_t r;

  // Valid route lines
  TEST_ASSERT_TRUE(route_parse("192.168.1.0/24 10.0.0.1 eth0", &r));

  TEST_ASSERT_EQUAL_UINT8(24, r.dest.prefix);

  TEST_ASSERT_EQUAL_STRING("eth0", r.dev);

  TEST_ASSERT_TRUE(route_parse("default direct eth1", &r));
  TEST_ASSERT_EQUAL_HEX32(0x00000000, r.dest.addr);
  TEST_ASSERT_EQUAL_UINT8(0, r.dest.prefix);
  TEST_ASSERT_EQUAL_UINT32(0, r.gateway);

  // Invalid route lines
  TEST_ASSERT_FALSE(route_parse(NULL, &r));
  TEST_ASSERT_FALSE(route_parse("   ", &r));                                    // Empty/whitespace only
  TEST_ASSERT_FALSE(route_parse("192.168.1.5/24 10.0.0.1 eth0", &r));           // host bits set in dest
  TEST_ASSERT_FALSE(route_parse("192.168.1.0/24 bad_gateway eth0", &r));        // bad gateway

  TEST_ASSERT_FALSE(route_parse("192.168.1.0/24 direct eth0\rname", &r));       // control whitespace inside device name
}

static void test_fwd_table_operations(void)
{
  fwd_table_t *table = table_create();
  TEST_ASSERT_NOT_NULL(table);
  TEST_ASSERT_EQUAL_INT(0, table_count(table));

  route_t r1 = {.dest = {0xC0A80100, 24}, .gateway = 0x0A000001, .dev = "eth0"};
  route_t r2 = {.dest = {0xC0A80000, 16}, .gateway = 0x0A000002, .dev = "eth1"};
  route_t r_default = {.dest = {0x00000000, 0}, .gateway = 0x0A000003, .dev = "eth2"};

  // Invalid table adds
  TEST_ASSERT_FALSE(table_add(NULL, &r1));
  TEST_ASSERT_FALSE(table_add(table, NULL));

  route_t bad_dest_route = {.dest = {0xC0A80105, 24}, .gateway = 0x0A000001, .dev = "eth0"};
  TEST_ASSERT_FALSE(table_add(table, &bad_dest_route)); // Non-zero host bits in dest

  // Valid adds
  TEST_ASSERT_TRUE(table_add(table, &r1));
  TEST_ASSERT_EQUAL_INT(1, table_count(table));

  // Duplicate add should fail
  TEST_ASSERT_FALSE(table_add(table, &r1));

  TEST_ASSERT_TRUE(table_add(table, &r2));
  TEST_ASSERT_TRUE(table_add(table, &r_default));
  TEST_ASSERT_EQUAL_INT(3, table_count(table));

  // Test capacity expansion by adding multiple routes to trigger realloc block
  for (int i = 1; i <= 10; i++)
  {
    route_t extra_r = {.dest = {(uint32_t)(0x0A000000 + (i << 16)), 16}, .gateway = 0, .dev = "eth3"};
    // Ensure network address representation is valid
    extra_r.dest.addr &= cidr_mask(extra_r.dest.prefix);
    // Avoid duplicate if i collides
    (void)table_add(table, &extra_r);
  }

  // Test Longest Prefix Match (table_lookup)
  const route_t *matched = table_lookup(table, 0xC0A8010A);
  TEST_ASSERT_NOT_NULL(matched);
  TEST_ASSERT_EQUAL_STRING("eth0", matched->dev);

  matched = table_lookup(table, 0xC0A80501);
  TEST_ASSERT_NOT_NULL(matched);
  TEST_ASSERT_EQUAL_STRING("eth1", matched->dev);

  matched = table_lookup(table, 0x7F000001);
  TEST_ASSERT_NOT_NULL(matched);
  TEST_ASSERT_EQUAL_STRING("eth2", matched->dev);

  // NULL table lookup
  TEST_ASSERT_NULL(table_lookup(NULL, 0xC0A8010A));

  // Test table_load and error handling
  fwd_table_t *loaded_table = table_create();
  size_t bad_line = 0;

  FILE *tmp = tmpfile();
  TEST_ASSERT_NOT_NULL(tmp);
  fprintf(tmp, "# Comment line\n");
  fprintf(tmp, "  \n"); // Blank line
  fprintf(tmp, "10.0.0.0/8 direct eth3\n");
  rewind(tmp);
  TEST_ASSERT_TRUE(table_load(loaded_table, tmp, &bad_line));
  TEST_ASSERT_EQUAL_INT(0, bad_line);
  TEST_ASSERT_EQUAL_INT(1, table_count(loaded_table));
  fclose(tmp);

  // Test table_load with invalid file / arguments
  TEST_ASSERT_FALSE(table_load(NULL, NULL, &bad_line));
  TEST_ASSERT_FALSE(table_load(loaded_table, NULL, &bad_line));

  // Test table_load with bad route line tracking
  FILE *bad_tmp = tmpfile();
  fprintf(bad_tmp, "192.168.1.0/24 10.0.0.1 eth0\n");
  fprintf(bad_tmp, "malformed_line_here\n");
  rewind(bad_tmp);

  size_t failed_line_num = 0;
  TEST_ASSERT_FALSE(table_load(loaded_table, bad_tmp, &failed_line_num));
  TEST_ASSERT_EQUAL_INT(2, failed_line_num);
  fclose(bad_tmp);

  // A directory stream fails when getline attempts to read it.
  FILE *error_stream = fopen(".", "r");
  TEST_ASSERT_NOT_NULL(error_stream);
  size_t io_error_line = 1;
  TEST_ASSERT_FALSE(table_load(loaded_table, error_stream, &io_error_line));
  TEST_ASSERT_EQUAL_INT(0, io_error_line);
  TEST_ASSERT_TRUE(ferror(error_stream));
  fclose(error_stream);

  // Clean up
  table_destroy(table);
  table_destroy(loaded_table);
  table_destroy(NULL); // Should safely handle NULL
}
```
